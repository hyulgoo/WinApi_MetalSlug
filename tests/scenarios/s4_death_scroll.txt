# MS-R16: 화면을 밀고 가다가 죽어도 몬스터가 배경에 붙어 있다
# 수정 전: 죽은 동안(2.7초)은 RenewalPos가 돌지 않아 카메라는 멈추는데 "카메라가 움직인다" 플래그가 남아

expect end created.CZombie_boy == 1
expect end deleted.CZombie_boy == 0
expect 4s player.anim == DEAD_SLIME.anim
expect end wrange.CZombie_boy.x <= 1
