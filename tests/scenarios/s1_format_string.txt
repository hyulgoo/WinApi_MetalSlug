# MS-R08: 이름에 %가 든 애니메이션을 저장하고 다시 불러올 수 있다
# 수정 전: 파일 이름·애니메이션 이름을 fwprintf_s의 포맷 문자열 자리에 넣어, "%n"이 서식으로
#          처리되며 CRT가 잘못된 매개변수로 프로세스를 끝낸다
# (Stage 9에서 MS-R20 누수를 고쳐 allowleaks를 뺐다. 버튼 위치는 새 패널 기준)
start ANIMEDITOR
frames 160
log 0
# 시트에서 프레임 하나를 드래그한다
mouse 5 100 100
hold 10 16 LBTN
mouse 13 200 200
expect 20 editor.frames == 1
# 패널의 Save 버튼(1145, 25)을 누르고, 대화상자에는 이름에 %가 든 경로를 준다
mouse 25 1145 25
tap 30 LBTN
dialog $out\hit100%n.anim
expect 40 errors == 0
expect 40 editor.name == hit100%n.anim
expect 40 animfile.$out\hit100%n.anim.name == hit100%n.anim
# New(955, 25)로 비운 뒤 Load(1050, 25)로 같은 파일을 다시 불러온다
mouse 45 955 25
tap 50 LBTN
expect 55 editor.frames == 0
mouse 60 1050 25
tap 65 LBTN
dialog $out\hit100%n.anim
expect 80 errors == 0
expect 80 editor.frames == 1
expect 80 editor.name == hit100%n.anim
expect 80 level == ANIMEDITOR
