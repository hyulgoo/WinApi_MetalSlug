# 수동 점검표(docs/manual-checklist.md) 흐름을 자동으로 한 바퀴 돈다
# 타이틀 → 스테이지 → 전투 → 사망 3회 → 게임오버 → 타이틀 → 다시 시작 → 에디터 왕복 → 다시 시작
start START
frames 3600
log 30
logmon 120
heapcheck
tap 20 RIGHT
expect 25 level == START
tap 40 C
expect 170 level == STAGE
expect 170 alive.CCQCArea == 1
# 진행하며 사격·폭탄·점프
hold 200 380 RIGHT
tap 230 C
tap 260 C
tap 300 Z
tap 330 X
# 가까이 붙은 좀비들에게 맞아 목숨을 모두 잃는다
spawn 400 ZB_BOY 300 450
spawn 400 ZB_GIRL 380 450
spawn 400 ZB_MAN 460 450
spawn 1100 ZB_BOY 260 450
spawn 1100 ZB_MAN 340 450
expect 2300 level == GAMEOVER
# 게임오버에서 C를 2초 눌러 타이틀로
hold 2400 2560 C
expect 2600 level == START
expect 2600 alive.CSelectAnim == 1
# 다시 시작 — 근접 영역이 있어야 한다(MS-R02)
tap 2620 C
expect 2750 level == STAGE
expect 2750 alive.CCQCArea == 1
# 타이틀로 돌아가 에디터 왕복
level 2800 START
tap 2830 F1
expect 2850 level == ANIMEDITOR
tap 2900 F4
expect 2920 level == START
tap 2950 F1
expect 2970 level == ANIMEDITOR
tap 3000 F4
tap 3040 C
expect 3200 level == STAGE
expect end errors == 0
